## 📖 FAT32 基础知识

### 1. 什么是 FAT？

**FAT** = File Allocation Table（文件分配表）

**核心思想：** 用一个表记录文件的"碎片"在哪里

```
想象一个仓库：
- 货物（文件）可能放在多个货架（簇）上
- FAT 表就是记录"这批货物在哪些货架"的清单
```

### 2. FAT32 vs FAT16 vs FAT12

| 特性 | FAT12 | FAT16 | FAT32 |
|------|-------|-------|-------|
| 簇编号位数 | 12位 | 16位 | 28位（32位中的28位）|
| 最大卷大小 | 32MB | 4GB | 2TB |
| 最大文件 | 32MB | 2GB | 4GB |
| 使用场景 | 软盘 | 老U盘 | 现代U盘、SD卡 |

### 3. FAT32 磁盘布局

```
┌─────────────────────────────────────────────┐
│ 保留区 (Reserved Sectors)                   │
│  ├── 引导扇区 (Boot Sector)                 │  ← 扇区 0
│  ├── FSInfo                                 │  ← 扇区 1
│  └── 备份引导扇区                           │  ← 扇区 6
├─────────────────────────────────────────────┤
│ FAT 1 (File Allocation Table 1)            │
├─────────────────────────────────────────────┤
│ FAT 2 (File Allocation Table 2，备份)      │
├─────────────────────────────────────────────┤
│ 数据区 (Data Region)                        │
│  ├── 根目录 (从簇 2 开始)                   │
│  ├── 子目录和文件                           │
│  └── ...                                    │
└─────────────────────────────────────────────┘
```

**关键概念：**
- **扇区（Sector）**：最小读写单位，通常 512 字节
- **簇（Cluster）**：分配单位，由多个扇区组成
- **FAT 表**：记录簇的使用情况和链接关系

## 🏗️ 核心数据结构

### 1. 引导扇区（Boot Sector）

```c
/*
 * FAT32 引导扇区结构（扇区 0）
 * 
 * 大小：512 字节
 * 作用：描述文件系统布局
 */
struct fat32_boot_sector {
    /* 通用 DOS 引导记录 */
    uint8_t  jump[3];              // 0x00: 跳转指令（EB 58 90）
    char     oem_name[8];          // 0x03: OEM 名称
    
    /* BIOS 参数块（BPB） */
    uint16_t bytes_per_sector;     // 0x0B: 每扇区字节数（512）
    uint8_t  sectors_per_cluster;  // 0x0D: 每簇扇区数（通常 8）
    uint16_t reserved_sectors;     // 0x0E: 保留扇区数（通常 32）
    uint8_t  num_fats;             // 0x10: FAT 表数量（通常 2）
    uint16_t root_entry_count;     // 0x11: 根目录项数（FAT32 为 0）
    uint16_t total_sectors_16;     // 0x13: 总扇区数（FAT32 为 0）
    uint8_t  media_type;           // 0x15: 媒体类型（0xF8）
    uint16_t fat_size_16;          // 0x16: FAT 大小（FAT32 为 0）
    uint16_t sectors_per_track;    // 0x18: 每磁道扇区数
    uint16_t num_heads;            // 0x1A: 磁头数
    uint32_t hidden_sectors;       // 0x1C: 隐藏扇区数
    uint32_t total_sectors_32;     // 0x20: 总扇区数
    
    /* FAT32 扩展 BPB */
    uint32_t fat_size_32;          // 0x24: FAT 大小（扇区数）
    uint16_t ext_flags;            // 0x28: 扩展标志
    uint16_t fs_version;           // 0x2A: 文件系统版本（0）
    uint32_t root_cluster;         // 0x2C: 根目录起始簇号（通常 2）
    uint16_t fs_info;              // 0x30: FSInfo 扇区号（通常 1）
    uint16_t backup_boot_sector;   // 0x32: 备份引导扇区（通常 6）
    uint8_t  reserved[12];         // 0x34: 保留
    uint8_t  drive_number;         // 0x40: 驱动器号（0x80）
    uint8_t  reserved1;            // 0x41: 保留
    uint8_t  boot_signature;       // 0x42: 启动签名（0x29）
    uint32_t volume_id;            // 0x43: 卷序列号
    char     volume_label[11];     // 0x47: 卷标
    char     fs_type[8];           // 0x52: 文件系统类型 "FAT32   "
    uint8_t  boot_code[420];       // 0x5A: 引导代码
    uint16_t boot_signature_end;   // 0x1FE: 签名（0xAA55）
} __attribute__((packed));
```

**关键字段解析：**

```c
// 扇区大小（通常 512 字节）
bytes_per_sector = 512;

// 每簇扇区数（通常 8，即 4KB 簇）
sectors_per_cluster = 8;

// 簇大小 = bytes_per_sector * sectors_per_cluster
cluster_size = 512 * 8 = 4096 字节 = 4KB

// FAT 表位置
fat_start_sector = reserved_sectors;

// 数据区位置
data_start_sector = reserved_sectors + (num_fats * fat_size_32);

// 根目录位置（簇号）
root_cluster = 2;  // 第一个可用簇
```

### 2. 目录项（Directory Entry）

```c
/*
 * FAT32 目录项（短文件名格式）
 * 
 * 大小：32 字节
 * 作用：描述一个文件或目录
 */
struct fat32_dir_entry {
    char     name[11];             // 0x00: 8.3 格式文件名
    uint8_t  attr;                 // 0x0B: 文件属性
    uint8_t  nt_reserved;          // 0x0C: 保留（Windows NT）
    uint8_t  create_time_tenth;    // 0x0D: 创建时间（十分之一秒）
    uint16_t create_time;          // 0x0E: 创建时间
    uint16_t create_date;          // 0x10: 创建日期
    uint16_t last_access_date;     // 0x12: 最后访问日期
    uint16_t first_cluster_high;   // 0x14: 起始簇号（高 16 位）
    uint16_t write_time;           // 0x16: 修改时间
    uint16_t write_date;           // 0x18: 修改日期
    uint16_t first_cluster_low;    // 0x1A: 起始簇号（低 16 位）
    uint32_t file_size;            // 0x1C: 文件大小
} __attribute__((packed));

/* 文件属性 */
#define ATTR_READ_ONLY   0x01      // 只读
#define ATTR_HIDDEN      0x02      // 隐藏
#define ATTR_SYSTEM      0x04      // 系统
#define ATTR_VOLUME_ID   0x08      // 卷标
#define ATTR_DIRECTORY   0x10      // 目录
#define ATTR_ARCHIVE     0x20      // 归档
#define ATTR_LONG_NAME   0x0F      // 长文件名（特殊）
```

**8.3 文件名格式：**

```
示例："HELLO   TXT"
     ├─────┬────┤
     8个字符  3个字符扩展名
     
- "HELLO.TXT"  → "HELLO   TXT"
- "test.c"     → "TEST    C  "
- "README.md"  → "README  MD "（截断）
```

**获取起始簇号：**

```c
uint32_t get_first_cluster(struct fat32_dir_entry *entry) {
    return ((uint32_t)entry->first_cluster_high << 16) | 
            entry->first_cluster_low;
}
```

### 3. 长文件名（LFN）

长文件名通过特殊的目录项实现：

```c
/*
 * 长文件名目录项
 * 
 * 特点：attr = ATTR_LONG_NAME (0x0F)
 */
struct fat32_lfn_entry {
    uint8_t  order;                // 序号（从后往前，最后一个 OR 0x40）
    uint16_t name1[5];             // 名称第 1-5 字符（Unicode）
    uint8_t  attr;                 // 属性（固定 0x0F）
    uint8_t  type;                 // 类型（固定 0）
    uint8_t  checksum;             // 短文件名校验和
    uint16_t name2[6];             // 名称第 6-11 字符
    uint16_t first_cluster_low;    // 固定 0
    uint16_t name3[2];             // 名称第 12-13 字符
} __attribute__((packed));
```

**长文件名示例：**

```
文件名："LongFileName.txt"

目录项排列：
┌──────────────────────────────────┐
│ LFN Entry (序号 0x42, "ext")     │  ← 最后一部分，序号 OR 0x40
├──────────────────────────────────┤
│ LFN Entry (序号 0x01, "LongFileN")│
├──────────────────────────────────┤
│ Short Entry ("LONGFI~1.TXT")     │  ← 对应的短文件名
└──────────────────────────────────┘
```

### 4. FAT 表

FAT 表记录簇的使用情况：

```c
/*
 * FAT 表项含义（28 位有效）
 */
#define FAT32_FREE       0x00000000  // 空闲簇
#define FAT32_RESERVED   0x00000001  // 保留
#define FAT32_BAD        0x0FFFFFF7  // 坏簇
#define FAT32_EOC_MIN    0x0FFFFFF8  // 簇链结束（最小值）
#define FAT32_EOC_MAX    0x0FFFFFFF  // 簇链结束（最大值）

/*
 * 簇链示例：
 * 
 * 文件占用簇：2 → 3 → 5 → 7 → EOF
 * 
 * FAT[2] = 3
 * FAT[3] = 5
 * FAT[5] = 7
 * FAT[7] = 0x0FFFFFFF  (EOF)
 */
```

**读取 FAT 表项：**

```c
uint32_t read_fat_entry(uint32_t cluster) {
    // FAT 表在磁盘上的位置
    uint32_t fat_offset = cluster * 4;  // 每个表项 4 字节
    uint32_t fat_sector = fat_start_sector + (fat_offset / bytes_per_sector);
    uint32_t entry_offset = fat_offset % bytes_per_sector;
    
    // 读取扇区
    uint8_t sector_buf[512];
    read_sector(fat_sector, sector_buf);
    
    // 提取表项（只用低 28 位）
    uint32_t entry = *(uint32_t*)(sector_buf + entry_offset);
    return entry & 0x0FFFFFFF;
}
```

### 阶段 1：挂载文件系统（第 1-2 天）

**目标：** 读取并解析引导扇区

```c
struct fat32_fs {
    struct vfs_superblock *sb;
    
    // 从引导扇区读取的信息
    uint32_t bytes_per_sector;
    uint32_t sectors_per_cluster;
    uint32_t cluster_size;
    uint32_t reserved_sectors;
    uint32_t num_fats;
    uint32_t fat_size;
    uint32_t root_cluster;
    uint32_t total_sectors;
    
    // 计算的偏移
    uint32_t fat_start_sector;
    uint32_t data_start_sector;
    uint32_t total_clusters;
    
    // 设备
    struct block_device *bdev;
};

int fat32_mount(struct block_device *bdev) {
    // 1. 读取引导扇区
    struct fat32_boot_sector bs;
    read_sector(bdev, 0, &bs);
    
    // 2. 验证签名
    if (bs.boot_signature_end != 0xAA55) {
        return -EINVAL;
    }
    
    // 3. 验证文件系统类型
    if (strncmp(bs.fs_type, "FAT32", 5) != 0) {
        return -EINVAL;
    }
    
    // 4. 提取参数
    struct fat32_fs *fs = kmalloc(sizeof(*fs));
    fs->bytes_per_sector = bs.bytes_per_sector;
    fs->sectors_per_cluster = bs.sectors_per_cluster;
    fs->cluster_size = bs.bytes_per_sector * bs.sectors_per_cluster;
    // ...
    
    // 5. 计算偏移
    fs->fat_start_sector = bs.reserved_sectors;
    fs->data_start_sector = bs.reserved_sectors + 
                            (bs.num_fats * bs.fat_size_32);
    
    return 0;
}
```

### 阶段 2：读取目录（第 3-4 天）

**目标：** 遍历目录，解析文件列表

```c
/*
 * 读取目录项
 */
int fat32_readdir(struct vfs_file *file) {
    struct fat32_fs *fs = get_fs(file);
    uint32_t cluster = get_dir_cluster(file);
    
    while (cluster < FAT32_EOC_MIN) {
        // 读取簇
        uint8_t *cluster_data = read_cluster(fs, cluster);
        
        // 遍历目录项（每簇可能有多个）
        int entries_per_cluster = fs->cluster_size / sizeof(struct fat32_dir_entry);
        
        for (int i = 0; i < entries_per_cluster; i++) {
            struct fat32_dir_entry *entry = 
                (struct fat32_dir_entry*)(cluster_data + i * 32);
            
            // 跳过空项
            if (entry->name[0] == 0x00) break;      // 结束
            if (entry->name[0] == 0xE5) continue;   // 已删除
            
            // 跳过长文件名项（暂时）
            if (entry->attr == ATTR_LONG_NAME) continue;
            
            // 跳过卷标
            if (entry->attr & ATTR_VOLUME_ID) continue;
            
            // 打印文件名
            char name[13];
            fat32_format_name(entry->name, name);
            kprintf("File: %s\n", name);
        }
        
        // 下一个簇
        cluster = read_fat_entry(fs, cluster);
    }
    
    return 0;
}
```

**格式化文件名：**

```c
void fat32_format_name(const char *fat_name, char *output) {
    // "HELLO   TXT" → "HELLO.TXT"
    int j = 0;
    
    // 处理主文件名（8 字符）
    for (int i = 0; i < 8 && fat_name[i] != ' '; i++) {
        output[j++] = fat_name[i];
    }
    
    // 处理扩展名（3 字符）
    if (fat_name[8] != ' ') {
        output[j++] = '.';
        for (int i = 8; i < 11 && fat_name[i] != ' '; i++) {
            output[j++] = fat_name[i];
        }
    }
    
    output[j] = '\0';
}
```

### 阶段 3：文件查找（第 5 天）

**目标：** 根据路径查找文件

```c
/*
 * 查找文件
 */
struct fat32_dir_entry *fat32_lookup(
    struct fat32_fs *fs,
    const char *path)
{
    // 1. 从根目录开始
    uint32_t cluster = fs->root_cluster;
    
    // 2. 分割路径
    char *path_copy = strdup(path);
    char *token = strtok(path_copy, "/");
    
    while (token) {
        // 在当前目录中查找
        struct fat32_dir_entry *entry = find_in_dir(fs, cluster, token);
        
        if (!entry) {
            kfree(path_copy);
            return NULL;  // 未找到
        }
        
        // 如果是最后一个部分，返回
        char *next = strtok(NULL, "/");
        if (!next) {
            kfree(path_copy);
            return entry;
        }
        
        // 否则，进入子目录
        if (!(entry->attr & ATTR_DIRECTORY)) {
            kfree(path_copy);
            return NULL;  // 不是目录
        }
        
        cluster = get_first_cluster(entry);
        token = next;
    }
    
    kfree(path_copy);
    return NULL;
}
```

### 阶段 4：文件读取（第 6-7 天）

**目标：** 读取文件内容

```c
/*
 * 读取文件
 */
int fat32_read_file(
    struct vfs_file *file,
    char *buf,
    size_t count)
{
    struct fat32_fs *fs = get_fs(file);
    struct fat32_dir_entry *entry = file->private_data;
    
    // 获取起始簇
    uint32_t cluster = get_first_cluster(entry);
    uint32_t offset = file->pos;
    uint32_t file_size = entry->file_size;
    
    // 调整读取大小
    if (offset >= file_size) return 0;  // EOF
    if (offset + count > file_size) {
        count = file_size - offset;
    }
    
    // 跳过开头的簇
    uint32_t cluster_offset = offset / fs->cluster_size;
    for (uint32_t i = 0; i < cluster_offset && cluster < FAT32_EOC_MIN; i++) {
        cluster = read_fat_entry(fs, cluster);
    }
    
    // 簇内偏移
    uint32_t offset_in_cluster = offset % fs->cluster_size;
    uint32_t bytes_read = 0;
    
    // 循环读取簇
    while (bytes_read < count && cluster < FAT32_EOC_MIN) {
        // 读取簇
        uint8_t *cluster_data = read_cluster(fs, cluster);
        
        // 计算本次读取大小
        uint32_t to_read = fs->cluster_size - offset_in_cluster;
        if (to_read > count - bytes_read) {
            to_read = count - bytes_read;
        }
        
        // 复制数据
        memcpy(buf + bytes_read, cluster_data + offset_in_cluster, to_read);
        bytes_read += to_read;
        offset_in_cluster = 0;  // 后续簇从头开始
        
        // 下一个簇
        cluster = read_fat_entry(fs, cluster);
    }
    
    file->pos += bytes_read;
    return bytes_read;
}
```

### 阶段 5：文件写入（第 8-10 天）

**目标：** 创建文件和写入数据

```c
/*
 * 分配新簇
 */
uint32_t fat32_alloc_cluster(struct fat32_fs *fs) {
    // 1. 扫描 FAT 表找空闲簇
    for (uint32_t cluster = 2; cluster < fs->total_clusters; cluster++) {
        uint32_t entry = read_fat_entry(fs, cluster);
        if (entry == FAT32_FREE) {
            // 2. 标记为已使用（EOF）
            write_fat_entry(fs, cluster, FAT32_EOC_MAX);
            return cluster;
        }
    }
    
    return 0;  // 磁盘已满
}

/*
 * 写入文件
 */
int fat32_write_file(
    struct vfs_file *file,
    const char *buf,
    size_t count)
{
    struct fat32_fs *fs = get_fs(file);
    struct fat32_dir_entry *entry = file->private_data;
    
    uint32_t cluster = get_first_cluster(entry);
    uint32_t offset = file->pos;
    
    // 如果是新文件，分配第一个簇
    if (cluster == 0) {
        cluster = fat32_alloc_cluster(fs);
        if (!cluster) return -ENOSPC;  // 磁盘已满
        
        entry->first_cluster_high = cluster >> 16;
        entry->first_cluster_low = cluster & 0xFFFF;
    }
    
    // 定位到offset所在的簇
    // ... (类似read的逻辑)
    
    uint32_t bytes_written = 0;
    
    while (bytes_written < count) {
        // 读取现有簇
        uint8_t *cluster_data = read_cluster(fs, cluster);
        
        // 修改数据
        uint32_t to_write = fs->cluster_size - offset_in_cluster;
        if (to_write > count - bytes_written) {
            to_write = count - bytes_written;
        }
        
        memcpy(cluster_data + offset_in_cluster, buf + bytes_written, to_write);
        
        // 写回簇
        write_cluster(fs, cluster, cluster_data);
        bytes_written += to_write;
        offset_in_cluster = 0;
        
        // 需要新簇？
        uint32_t next = read_fat_entry(fs, cluster);
        if (next >= FAT32_EOC_MIN) {
            if (bytes_written < count) {
                // 分配新簇
                next = fat32_alloc_cluster(fs);
                if (!next) break;  // 磁盘已满
                
                // 链接簇
                write_fat_entry(fs, cluster, next);
            }
        }
        
        cluster = next;
    }
    
    // 更新文件大小
    if (file->pos + bytes_written > entry->file_size) {
        entry->file_size = file->pos + bytes_written;
        update_dir_entry(fs, entry);
    }
    
    file->pos += bytes_written;
    return bytes_written;
}
```

### 阶段 6：长文件名支持（第 11-12 天）

**目标：** 解析和创建长文件名

```c
/*
 * 读取长文件名
 */
int fat32_read_lfn(
    struct fat32_dir_entry *entries,
    int count,
    char *name_out)
{
    char lfn_buf[256] = {0};
    int lfn_pos = 0;
    
    // 从后往前读取 LFN 项
    for (int i = 0; i < count; i++) {
        if (entries[i].attr != ATTR_LONG_NAME) break;
        
        struct fat32_lfn_entry *lfn = (struct fat32_lfn_entry*)&entries[i];
        
        // 提取名称片段
        for (int j = 0; j < 5; j++) {
            if (lfn->name1[j] == 0 || lfn->name1[j] == 0xFFFF) break;
            lfn_buf[lfn_pos++] = lfn->name1[j] & 0xFF;
        }
        for (int j = 0; j < 6; j++) {
            if (lfn->name2[j] == 0 || lfn->name2[j] == 0xFFFF) break;
            lfn_buf[lfn_pos++] = lfn->name2[j] & 0xFF;
        }
        for (int j = 0; j < 2; j++) {
            if (lfn->name3[j] == 0 || lfn->name3[j] == 0xFFFF) break;
            lfn_buf[lfn_pos++] = lfn->name3[j] & 0xFF;
        }
    }
    
    strcpy(name_out, lfn_buf);
    return lfn_pos;
}
```

## 📊 性能优化

### 1. FAT 表缓存

```c
struct fat_cache {
    uint32_t cluster;
    uint32_t next_cluster;
    bool valid;
};

struct fat_cache fat_cache[256];  // 缓存 256 个表项

uint32_t read_fat_entry_cached(struct fat32_fs *fs, uint32_t cluster) {
    int idx = cluster % 256;
    
    if (fat_cache[idx].valid && fat_cache[idx].cluster == cluster) {
        return fat_cache[idx].next_cluster;  // 命中缓存
    }
    
    // 未命中，读取
    uint32_t next = read_fat_entry(fs, cluster);
    
    // 更新缓存
    fat_cache[idx].cluster = cluster;
    fat_cache[idx].next_cluster = next;
    fat_cache[idx].valid = true;
    
    return next;
}
```

## 💡 常见问题

### Q1: 为什么簇号从 2 开始？

**A:** 簇号 0 和 1 保留：
- 簇 0：表示空闲或无效
- 簇 1：保留
- 簇 2：第一个可用簇（通常是根目录）

### Q2: FAT32 为什么叫 FAT32？

**A:** 因为 FAT 表项使用 32 位（实际只用 28 位）

```bash
# 1. 格式化 U 盘为 FAT32
```

### 实现中遇到的问题和解决方案

#### 问题1：磁盘大小导致 FAT16 而非 FAT32

**问题：** 最初创建 16MB 测试磁盘，mkfs.vfat 创建的是 FAT16

**症状：**
```
[FAT32] Sectors per cluster: 1          ← FAT16 特征
[FAT32] total_sectors_16 should be 0    ← 验证失败
```

**解决方案：** 将磁盘大小增加到 **64MB 或更大**
```bash
DISK_SIZE_MB=128  # FAT32 需要足够大的磁盘
```

**学习要点：** FAT32 有最小磁盘大小要求（通常 33MB+）

#### 问题2：QEMU IDE Primary Slave 检测失败

**问题：** Primary Slave (hdb) 完全不响应（状态 0x00）

**原因：** QEMU 的 `-drive` 参数默认行为问题

**解决方案1：** 使用老式 QEMU 语法
```bash
-hda eduos.img        # Primary Master
-hdb fat32_test.img   # Primary Slave ✅ 成功！
```

**解决方案2：** Linux 风格探测降级
```c
// 如果 IDENTIFY 失败，尝试直接读取扇区
if (!ide_identify(dev)) {
    ide_probe_read(dev);  // 发送 READ 命令代替 IDENTIFY
}
```

**学习要点：** 真实硬件不完全符合规范时，Linux 使用多种降级策略

#### 问题3：Secondary IDE 控制器错误

**问题：** Secondary Master (hdc) 返回错误（状态 0x41）

**症状：**
```
[IDE] Initial status: 0x41    ← ERR 标志设置
[IDE] Probe timeout           ← 无法读取数据
```

**原因：** QEMU Secondary IDE 控制器模拟不完整

**解决方案：** 放弃 Secondary，专注 Primary
```c
// Primary Master (hda) - EduOS 系统盘  ✅
// Primary Slave (hdb)  - FAT32 测试盘 ✅
// Secondary 不使用
```

#### 问题4：字符串格式化

**问题：** 简化的 `kprintf` 不支持 `%.8s` 格式

**症状：**
```c
kprintf("FS Type: %.8s\n", bs.fs_type);  // 输出乱码
```

**解决方案：** 手动逐字符打印
```c
kprintf("FS Type: [%c%c%c%c%c%c%c%c]\n", 
        bs.fs_type[0], bs.fs_type[1], ...);
```

**学习要点：** 裸机 kprintf 功能有限，需要手动处理

#### 问题5：浮点数打印

**问题：** 内核禁用浮点运算（`-mno-sse -mno-mmx`）

**解决方案：** 改用整数运算
```c
// ❌ 错误：kprintf("Size: %.2f MB\n", sectors * 512.0 / 1024 / 1024);
// ✅ 正确：
uint32_t size_mb = (sectors * 512) / (1024 * 1024);
kprintf("Size: %u MB\n", size_mb);
```
