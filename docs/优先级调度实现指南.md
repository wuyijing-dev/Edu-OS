# 优先级调度实现指南

## 📚 概述

本文档详细介绍如何在EduOS中实现生产级的优先级调度系统，包括数据结构设计、算法实现和防饿死机制。

---

## 🎯 设计目标

### 功能需求
1. ✅ 支持32个优先级级别（0-31，0最高）
2. ✅ 高优先级进程优先执行
3. ✅ 同优先级进程使用时间片轮转
4. ✅ 防止低优先级进程饿死（优先级老化）
5. ✅ O(1)时间复杂度的进程选择

### 性能指标
- 进程选择：O(1)
- 进程入队：O(1)
- 进程出队：O(1)
- 内存开销：最小化

## 🏗️ 数据结构设计

### 1. 优先级队列结构

```c
#define MAX_PRIORITIES 32  /* 32个优先级级别 */

/* 单个优先级队列 */
struct priority_queue {
    struct process *head;        /* 队列头 */
    struct process *tail;        /* 队列尾 */
    uint32_t count;              /* 进程数量 */
};

/* 优先级调度器 */
struct priority_scheduler {
    struct priority_queue queues[MAX_PRIORITIES];  /* 32个队列 */
    uint32_t bitmap;             /* 位图：标记哪些队列非空 */
    uint32_t total_count;        /* 总进程数 */
    uint64_t last_aging_time;    /* 上次老化时间 */
};
```

**位图的作用**：

```
bitmap是一个32位整数，每一位代表一个优先级队列是否有进程：

位31 位30 ... 位1 位0
 │    │        │   │
 └────┴────────┴───┴─→ 各个优先级队列

bitmap = 0x00000005 = 0b00000000000000000000000000000101
表示：优先级0和优先级2的队列有进程

使用位运算快速找到最高优先级：
int highest = __builtin_ffs(bitmap) - 1;  // O(1)时间
```

### 2. 进程优先级字段

在PCB中添加：

```c
struct process {
    /* ... 其他字段 ... */
    
    /* 优先级相关 */
    int priority;                /* 当前优先级（0-31） */
    int base_priority;           /* 基础优先级（不变） */
    uint64_t wait_time;          /* 等待时间（用于老化） */
    int boost_count;             /* 被提升次数 */
};
```

## 💻 核心算法实现

### 算法1：初始化

```c
static struct priority_scheduler priority_sched;

void priority_scheduler_init(void)
{
    /* 清空所有队列 */
    for (int i = 0; i < MAX_PRIORITIES; i++) {
        priority_sched.queues[i].head = NULL;
        priority_sched.queues[i].tail = NULL;
        priority_sched.queues[i].count = 0;
    }
    
    priority_sched.bitmap = 0;
    priority_sched.total_count = 0;
    priority_sched.last_aging_time = timer_get_ticks();
    
    kprintf("[PRIORITY] Priority Scheduler initialized (32 levels)\n");
}
```

### 算法2：进程入队（O(1)）

```c
void priority_enqueue(struct process *proc)
{
    if (!proc) return;
    
    int pri = proc->priority;
    
    /* 验证优先级范围 */
    if (pri < 0) pri = 0;
    if (pri >= MAX_PRIORITIES) pri = MAX_PRIORITIES - 1;
    proc->priority = pri;
    
    struct priority_queue *queue = &priority_sched.queues[pri];
    
    /* 加入队列尾部 */
    proc->next = NULL;
    proc->prev = queue->tail;
    
    if (queue->tail) {
        queue->tail->next = proc;
    } else {
        queue->head = proc;
    }
    queue->tail = proc;
    
    /* 更新计数 */
    queue->count++;
    priority_sched.total_count++;
    
    /* 设置位图 */
    priority_sched.bitmap |= (1 << pri);
    
    /* 记录入队时间 */
    proc->wait_time = timer_get_ticks();
}
```

### 算法3：进程出队（O(1)）

```c
void priority_dequeue(struct process *proc)
{
    if (!proc) return;
    
    int pri = proc->priority;
    if (pri < 0 || pri >= MAX_PRIORITIES) return;
    
    struct priority_queue *queue = &priority_sched.queues[pri];
    
    /* 从链表中移除 */
    if (proc->prev) {
        proc->prev->next = proc->next;
    } else {
        queue->head = proc->next;
    }
    
    if (proc->next) {
        proc->next->prev = proc->prev;
    } else {
        queue->tail = proc->prev;
    }
    
    proc->next = NULL;
    proc->prev = NULL;
    
    /* 更新计数 */
    queue->count--;
    priority_sched.total_count--;
    
    /* 如果队列空了，清除位图 */
    if (queue->count == 0) {
        priority_sched.bitmap &= ~(1 << pri);
    }
}
```

### 算法4：选择下一个进程（O(1)）

```c
struct process *priority_pick_next(void)
{
    /* 没有就绪进程 */
    if (priority_sched.bitmap == 0) {
        return NULL;
    }
    
    /* 使用位运算找到最高优先级（最低位） */
    int highest_pri = __builtin_ffs(priority_sched.bitmap) - 1;
    
    /* 从该优先级队列取出第一个进程 */
    struct priority_queue *queue = &priority_sched.queues[highest_pri];
    struct process *next = queue->head;
    
    if (!next) {
        /* 不应该发生：位图说有，但队列为空 */
        kprintf("[PRIORITY] WARNING: bitmap inconsistency at level %d\n", 
                highest_pri);
        priority_sched.bitmap &= ~(1 << highest_pri);
        return priority_pick_next();  /* 递归重试 */
    }
    
    /* 从队列中移除 */
    priority_dequeue(next);
    
    return next;
}
```

**__builtin_ffs说明**：

```c
/* 
 * __builtin_ffs(x) - Find First Set
 * 返回x中第一个为1的位的位置（从1开始计数）
 * 如果x=0，返回0
 * 
 * 示例：
 * __builtin_ffs(0b00000101) = 1  (第1位是1)
 * __builtin_ffs(0b00001000) = 4  (第4位是1)
 * __builtin_ffs(0) = 0
 */

/* 因此我们需要减1来得到0-based的索引 */
int pos = __builtin_ffs(bitmap) - 1;
```

### 算法5：优先级老化（防饿死）

```c
#define AGING_INTERVAL 100      /* 每100 ticks检查一次 */
#define AGING_THRESHOLD 200     /* 等待超过200 ticks就提升 */
#define AGING_BOOST 2           /* 每次提升2个级别 */

void priority_aging(void)
{
    uint64_t current_time = timer_get_ticks();
    
    /* 检查是否到了老化时间 */
    if (current_time - priority_sched.last_aging_time < AGING_INTERVAL) {
        return;
    }
    
    priority_sched.last_aging_time = current_time;
    
    /* 遍历所有优先级队列（从低到高） */
    for (int pri = MAX_PRIORITIES - 1; pri > 0; pri--) {
        struct priority_queue *queue = &priority_sched.queues[pri];
        struct process *proc = queue->head;
        
        while (proc) {
            struct process *next = proc->next;  /* 保存next，因为可能会移动 */
            
            /* 检查等待时间 */
            uint64_t wait_time = current_time - proc->wait_time;
            
            if (wait_time > AGING_THRESHOLD) {
                /* 计算新优先级 */
                int new_pri = pri - AGING_BOOST;
                if (new_pri < 0) new_pri = 0;
                
                if (new_pri != pri) {
                    /* 从当前队列移除 */
                    priority_dequeue(proc);
                    
                    /* 修改优先级 */
                    proc->priority = new_pri;
                    proc->boost_count++;
                    
                    /* 加入新队列 */
                    priority_enqueue(proc);
                    
                    /* 打印调试信息（可选） */
                    // kprintf("[AGING] Process %s (PID %u): %d -> %d\n",
                    //         proc->name, proc->pid, pri, new_pri);
                }
            }
            
            proc = next;
        }
    }
}
```

**老化算法流程图**：

```
定时检查（每100 ticks）
    │
    ├─→ 扫描所有队列
    │       │
    │       ├─→ 检查每个进程的等待时间
    │       │       │
    │       │       ├─→ 等待 > 200 ticks?
    │       │       │       │
    │       │       │       Yes → 提升优先级（-2）
    │       │       │       │
    │       │       │       No → 保持不变
    │       │       │
    │       │       └─→ 下一个进程
    │       │
    │       └─→ 下一个队列
    │
    └─→ 完成
```

### 算法6：调度主循环

```c
void priority_schedule(void)
{
    /* 执行优先级老化 */
    priority_aging();
    
    struct process *prev = current_process;
    struct process *next = priority_pick_next();
    
    if (!next) {
        /* 没有就绪进程，运行idle */
        next = idle_process;
    }
    
    /* 处理当前进程 */
    if (prev && prev->state == PROCESS_STATE_RUNNING) {
        if (prev->time_used >= prev->time_slice) {
            /* 时间片用完，变为就绪 */
            prev->state = PROCESS_STATE_READY;
            prev->time_used = 0;
            priority_enqueue(prev);
        } else {
            /* 主动让出或被抢占，保持优先级不变 */
            prev->state = PROCESS_STATE_READY;
            priority_enqueue(prev);
        }
    }
    
    /* 启动新进程 */
    next->state = PROCESS_STATE_RUNNING;
    next->time_used = 0;
    current_process = next;
    
    /* 执行上下文切换 */
    context_switch(prev, next);
}
```

## 📊 性能分析

### 时间复杂度

| 操作 | 复杂度 | 说明 |
|------|--------|------|
| 入队 | O(1) | 直接加入队列尾部 |
| 出队 | O(1) | 直接从队列移除 |
| 选择下一个 | O(1) | 使用位图快速定位 |
| 老化 | O(n) | 需要遍历所有进程，但定期执行 |

### 空间复杂度

```
结构体大小：
- priority_scheduler: ~400字节
  - 32个队列 × (head+tail+count)
  - bitmap: 4字节
  - 统计信息: ~16字节

每个进程额外开销：
- priority: 4字节
- base_priority: 4字节
- wait_time: 8字节
- boost_count: 4字节
总计：20字节/进程
```

### 优化建议

1. **位图优化**
```c
/* 使用内联汇编的bsf指令（更快） */
static inline int find_first_bit(uint32_t bitmap)
{
    int result;
    asm volatile("bsf %1, %0" : "=r"(result) : "r"(bitmap));
    return result;
}
```

2. **缓存友好**
```c
/* 将最近使用的队列索引缓存 */
static int last_priority = -1;

if (last_priority >= 0 && 
    priority_sched.queues[last_priority].count > 0) {
    /* 快速路径：上次的优先级还有进程 */
    return pick_from_queue(last_priority);
}
```

## 🐛 常见问题

### 问题1：饿死问题

**症状**：低优先级进程永远得不到CPU

**解决**：
```c
/* 实现优先级老化 */
1. 定期检查等待时间
2. 等待过久的进程提升优先级
3. 防止提升过度（设置最高限制）
```

### 问题2：优先级反转

**症状**：高优先级进程等待低优先级进程释放资源

**解决**：
```c
/* 优先级继承协议 */
当高优先级进程P1等待低优先级进程P2持有的资源时：
1. 临时提升P2的优先级到P1的级别
2. P2释放资源后，恢复原优先级
3. P1获得资源，继续执行
```

### 问题3：位图不一致

**症状**：位图说有进程，但队列为空

**解决**：
```c
/* 防御性编程 */
struct process *next = queue->head;
if (!next) {
    /* 修复位图 */
    bitmap &= ~(1 << priority);
    /* 记录错误 */
    kprintf("[ERROR] Bitmap inconsistency\n");
    /* 重试 */
    return pick_next();
}
```
